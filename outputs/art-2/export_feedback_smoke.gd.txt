extends SceneTree

func _initialize() -> void:
	call_deferred("_run")

func _run() -> void:
	var app: Control = (load("res://scenes/main.tscn") as PackedScene).instantiate() as Control
	root.add_child(app); await process_frame; await process_frame
	app.call("_start_survival_tissues", false); await process_frame; await process_frame
	var life: Sm2LifeScreen = app.find_child("LifeScreen", true, false) as Sm2LifeScreen
	if life == null: quit(1); return
	var s: Sm2JourneySession = life.session as Sm2JourneySession
	for index: int in 8:
		if not s.act(s.command("practice",s.world.hero_id(),"p5:activity.psionics")).ok: quit(2); return
	for node: String in ["p5:node.impulse","p5:node.shield"]:
		if not s.act(s.command("buy_node",s.world.hero_id(),node)).ok: quit(3); return
	if not s.act(s.command("travel",0,"ruins")).ok: quit(4); return
	life.redraw(); await process_frame; await process_frame
	(app.find_child("WorldBattle",true,false) as Button).pressed.emit(); await process_frame; await process_frame
	var screen: Sm2BattleScreen = app.find_child("BattleScreen",true,false) as Sm2BattleScreen
	if screen == null or screen.board.art == null: quit(5); return
	screen.auto_advance = false
	if screen.board.art.sprites.size() != 17: quit(6); return
	if not screen._execute(screen._command("move","",0,Vector2i(2,1))).accepted: quit(7); return
	if screen.board.motion.cues.is_empty() or screen.board.motion.cues[0].kind != "moved": quit(8); return
	if not screen._execute(screen._command("use_ability","p5:ability.shield",1)).accepted: quit(9); return
	if screen._actor(1).barrier.remaining != 18: quit(10); return
	screen._save(); var saved: String = s.state_hash(); screen._load()
	if s.state_hash() != saved or not screen.board.motion.cues.is_empty(): quit(11); return
	print("SM2_ART2_PACK_OK: movement, shield, 17 sprite regions, exact save/load")
	quit(0)
