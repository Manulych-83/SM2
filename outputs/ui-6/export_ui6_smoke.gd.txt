extends SceneTree
func _initialize() -> void: call_deferred("_run")
func _run() -> void:
	if ProjectSettings.get_setting("application/config/version")!="0.8.12-ui.6": quit(1); return
	if root.mode!=Window.MODE_EXCLUSIVE_FULLSCREEN or root.content_scale_size!=Vector2i(2560,1440): quit(2); return
	var app: Control=(load("res://scenes/main.tscn") as PackedScene).instantiate() as Control; root.add_child(app)
	await process_frame; await process_frame
	app.call("_start_survival_tissues",false); await process_frame; await process_frame
	var life: Sm2LifeScreen=app.find_child("LifeScreen",true,false) as Sm2LifeScreen
	if life==null: quit(3); return
	var s: Sm2JourneySession=life.session as Sm2JourneySession
	for index: int in 8:
		if not s.act(s.command("practice",2,"p5:activity.psionics")).ok: quit(4); return
	life.redraw(); await process_frame; await process_frame
	(app.find_child("Guide_development",true,false) as Button).pressed.emit(); await process_frame; await process_frame
	var hero: Sm2HeroDevelopmentScreen=app.find_child("HeroDevelopmentScreen",true,false) as Sm2HeroDevelopmentScreen
	if hero==null: quit(5); return
	hero._kind="skill"; hero._select("p1:skill.psionics",""); await process_frame; await process_frame
	if hero._sources.practice.is_empty(): quit(6); return
	var buy: Button=hero.find_child(Sm2HeroDevelopmentScreen.control_id("HeroBuy","p5:node.impulse"),true,false) as Button
	if buy==null or buy.disabled: quit(7); return
	buy.pressed.emit(); await process_frame; await process_frame
	if s.world.bodies[2].progress.tracks["p1:skill.psionics"].spent!=100 or s.world.bodies[2].progress.tracks["p1:skill.psionics"].earned!=200: quit(8); return
	await RenderingServer.frame_post_draw
	var picture: Image=root.get_texture().get_image()
	if picture.is_empty() or picture.save_png("E:/GPT/SM2/outputs/ui-6/pack-development.png")!=OK: quit(9); return
	(hero.find_child("HeroDevelopmentCompanion",true,false) as Button).pressed.emit(); await process_frame; await process_frame
	if hero.find_children("HeroBuy_*","Button",true,false).size()!=0 or hero._sources.companion.attributes.size()!=8: quit(10); return
	if not s.save_game().ok: quit(11); return
	var before: String=s.state_hash()
	if not s.load_game().ok or s.state_hash()!=before: quit(12); return
	print("SM2_UI6_PACK_OK: QHD development, XP sources, actual node purchase, separate companion view and exact save/load")
	quit(0)
