extends SceneTree
func _initialize() -> void: call_deferred("run")
func frames() -> void:
	for index in 6: await process_frame
func run() -> void:
	if ProjectSettings.get_setting("application/config/version")!="0.8.19-ui.13": quit(1); return
	var app: Control=(load("res://scenes/main.tscn") as PackedScene).instantiate() as Control; root.add_child(app); await frames()
	if "--verify" in OS.get_cmdline_user_args():
		if root.mode!=Window.MODE_WINDOWED or root.size!=Vector2i(1920,1080): quit(2); return
		app._open_campaign(0,true); await frames()
		if app._life.state_hash()!=FileAccess.get_file_as_string("E:/GPT/SM2/outputs/ui-13/pack-world-hash.txt"): quit(3); return
		print("SM2_UI13_RESTART_OK: persisted display and unchanged saved campaign across processes"); quit(0); return
	app._open_campaign(0,false); await frames()
	var session: Sm2JourneySession=app._life as Sm2JourneySession
	if not session.save_game().ok: quit(4); return
	var before: String=session.state_hash()
	(app.find_child("CampSettings",true,false) as Button).pressed.emit(); await frames()
	var settings: Sm2SettingsScreen=app.find_child("SettingsScreen",true,false) as Sm2SettingsScreen
	if settings==null: quit(5); return
	await RenderingServer.frame_post_draw
	if root.get_texture().get_image().save_png("E:/GPT/SM2/outputs/ui-13/pack-settings.png")!=OK: quit(6); return
	settings.draft={"mode":"windowed","resolution":"1920x1080","vsync":false}; settings.synchronize(); settings.preview(); await frames()
	if root.mode!=Window.MODE_WINDOWED or root.size!=Vector2i(1920,1080): quit(7); return
	settings.confirm(); await frames()
	if not Sm2DisplayPreferences.new().read().exists: quit(8); return
	settings.leave(); await frames()
	if app._life!=session or session.state_hash()!=before: quit(9); return
	var file: FileAccess=FileAccess.open("E:/GPT/SM2/outputs/ui-13/pack-world-hash.txt",FileAccess.WRITE); file.store_string(before); file.close()
	print("SM2_UI13_PACK_OK: packaged settings controls and display confirmation, live campaign preserved")
	quit(0)
